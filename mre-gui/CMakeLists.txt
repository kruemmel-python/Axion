cmake_minimum_required(VERSION 3.20)
project(MRE_GUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(MRE_Research_GUI WIN32 main.cpp)
target_compile_definitions(MRE_Research_GUI PRIVATE UNICODE _UNICODE)
target_link_libraries(MRE_Research_GUI PRIVATE comctl32 shell32 comdlg32)
if(MSVC)
  target_compile_options(MRE_Research_GUI PRIVATE /W4 /WX /utf-8)
endif()
